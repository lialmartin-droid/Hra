function attackRegion(id){
  const r=game.regions[id];
  if(!isFrontier(id)){log("Nemůžeš útočit na území, které nesousedí s tvojí říší.");render();return}
  const cost=Math.max(15,Math.floor(r.power*.32));
  if(game.army < r.power){
    game.army=Math.max(0,game.army-cost);
    log("Prohrál jsi bitvu o "+r.name+". Ztratil jsi část armády.");
    render();saveGame();return;
  }
  game.army-=cost;
  if(r.owner!=="neutral" && game.factions[r.owner]){
    game.factions[r.owner].relation-=35;
    game.factions[r.owner].ally=false;
  }
  r.owner="player";
  r.power=Math.floor(r.power*1.08);
  log("Dobyl jsi území "+r.name+". Spadá pod vládu Gariel.");
  game.selected=id;
  checkVictory();
  render();saveGame();
}

function trainArmy(){
  const g=80,w=45;
  if(game.gold<g || game.wood<w){log("Nemáš dost surovin na výcvik armády.");render();return}
  game.gold-=g;game.wood-=w;game.army+=35;
  log("Vytrénoval jsi nové jednotky pro armádu Gariel.");
  render();saveGame();
}

function proposeAlliance(id){
  const f=game.factions[id];
  if(!f){return}
  const sharedBorder=game.regions.some(r=>r.owner==="player" && r.neighbors.some(n=>game.regions[n].owner===id));
  const benefit = sharedBorder ? 25 : 0;
  const chance = f.relation + benefit + playerRegions().length*2 - owned(id).length;
  if(chance>=25){
    f.ally=true; f.relation+=20;
    log("Vznikla aliance s frakcí "+f.name+".");
  }else{
    f.relation-=5;
    log(f.name+" odmítá alianci. Musí z toho mít větší výhodu.");
  }
  render();saveGame();
}

function declareWar(id){
  const f=game.factions[id];
  if(!f)return;
  f.ally=false;f.relation=-100;
  log("Vyhlásil jsi válku frakci "+f.name+".");
  render();saveGame();
}

function aiExpand(id){
  const f=game.factions[id];
  if(!f || owned(id).length===0)return;
  const options=[];
  owned(id).forEach(r=>r.neighbors.forEach(n=>{
    const target=game.regions[n];
    if(target.owner!==id && target.owner!=="player" && !(game.factions[target.owner]||{}).ally){
      options.push(target);
    }
  }));
  if(options.length===0)return;
  const target=options[Math.floor(Math.random()*options.length)];
  if(f.army>target.power){
    f.army-=Math.floor(target.power*.25);
    target.owner=id;
    log(f.name+" obsadili území "+target.name+".");
  }
}

function aiAttackPlayer(id){
  const f=game.factions[id];
  if(!f || f.ally)return;
  const options=[];
  owned(id).forEach(r=>r.neighbors.forEach(n=>{
    const target=game.regions[n];
    if(target.owner==="player") options.push(target);
  }));
  if(options.length===0)return;
  const target=options[Math.floor(Math.random()*options.length)];
  if(f.army>target.power+game.army*.25){
    f.army-=Math.floor(target.power*.2);
    game.army=Math.max(0,game.army-25);
    target.owner=id;
    log(f.name+" dobyli tvoje území "+target.name+"!");
  }else{
    f.army=Math.max(0,f.army-35);
    log("Odrazil jsi útok frakce "+f.name+" na "+target.name+".");
  }
}

function aiTurn(){
  game.tick++;
  const inc=incomeFor("player");
  game.gold+=inc.gold;game.wood+=inc.wood;

  ["red","green","purple"].forEach(id=>{
    const f=game.factions[id];
    const incF=incomeFor(id);
    f.army += Math.floor((incF.gold+incF.wood)/7)+8;
    if(Math.random()<0.35) aiExpand(id);
    if(Math.random()<0.18) aiAttackPlayer(id);
    if(Math.random()<0.12 && !f.ally) f.relation += Math.floor(Math.random()*7)-3;
  });
  render();saveGame();
}

function checkVictory(){
  const nonPlayer=game.regions.filter(r=>r.owner!=="player" && r.owner!=="neutral").length;
  if(nonPlayer===0) log("Gariel ovládla všechny velké frakce světa.");
}

function saveGame(manual=false){
  localStorage.setItem(SAVE_KEY,JSON.stringify(game));
  if(manual){log("Hra uložena.");render();}
}
function loadGame(){
  const raw=localStorage.getItem(SAVE_KEY);
  if(raw){
    try{game=JSON.parse(raw)}catch(e){}
  }else{
    log("Začínáš jen s malou nekropolí. Dobývej sousední území a rozšiř říši.");
  }
  render();
  setTimeout(()=>{document.getElementById("viewport").scrollTo(80,80)},100);
}
function resetGame(){
  if(confirm("Opravdu začít novou hru?")){
    localStorage.removeItem(SAVE_KEY);
    location.reload();
  }
}
